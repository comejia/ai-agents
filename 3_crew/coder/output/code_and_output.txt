# Calcular los primeros 1000 términos de la serie 1 - 1/3 + 1/5 - 1/7 + ...

# Inicialización de la suma
suma = 0.0

# Bucle para calcular los primeros 1000 términos
for i in range(1000):
    termino = (-1)**i / (2*i + 1)  # Término de la serie
    suma += termino  # Sumar el término a la suma

# Multiplicar la suma por 4
resultado = suma * 4

# Mostrar el resultado
print(resultado)

# Salida esperada: 3.141592653589793
